### Set

用set在线log时间复杂度内求中位数

[K. Rainbow Subarray](https://codeforces.com/gym/104901/problem/K)

```c++
#include <algorithm>
#include <cmath>
#include <cstring>
#include <iostream>
#include <set>
#include <vector>

#define int long long
const int N = 5e5;
int mapp[N + 2];
struct node {
    int v;
    node(int x) {
        v = x;
    }
    bool operator<(const node &t) const {
        return v < t.v;
    }
};  // 其实感觉直接用int也没啥不行的
int n, m, sum1, sum2;  // 分别表示前半段 and 后半段的和
std::multiset<node> pre, pos;  // 分别表示前半段和后半段

int calc() {  // 这个函数与求解中位数无关，主要是为了解决把这一段所有的值调整到中位数上所需要的代价，（和具体题目有关）
    int median = pos.begin()->v;  // 这个就是中位数
    return (sum2 - median * (int)pos.size() + median * (int)pre.size() - sum1);
}

inline std::multiset<node>::iterator lst(std::multiset<node> &x) {
    std::multiset<node>::iterator res = x.end();  // 注意pre不能为空，这个一定要注意，否res--会出错
    res--;
    return res;
}
void adj() {  // 调整函数，目的是为了保证两段长度差不超过1且后半段较长
    if (pos.size() >= pre.size() + 2) {
        sum1 += pos.begin()->v;
        sum2 -= pos.begin()->v;
        pre.insert(pos.begin()->v);
        pos.erase(pos.begin());
        return;
    }
    if (pos.size() >= pre.size()) return;
    sum1 -= lst(pre)->v;
    sum2 += lst(pre)->v;
    pos.insert(lst(pre)->v);
    pre.erase(lst(pre));
}

void solve() {
    pre.clear(), pos.clear(), sum1 = 0, sum2 = 0;
    std::cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        std::cin >> mapp[i];
        mapp[i] -= i;
    }

    //   要维护pos始终比pre长
    int ans = 0;
    for (int h = 1, t = 1; h <= n; h++) {
        node ls1(mapp[h]);
        if (mapp[h] >= pos.begin()->v) {
            pos.insert(mapp[h]);
            sum2 += mapp[h];
        } else {
            pre.insert(mapp[h]);
            sum1 += mapp[h];
        }
        adj();
        while (calc() > m && t <= h) {
            node ls2(mapp[t]);
            if ((pos.begin()->v) <= ls2.v) {
                sum2 -= mapp[t];
                pos.erase(pos.find(ls2));
                adj();
            } else {
                sum1 -= mapp[t];
                pre.erase(pre.find(ls2));
                adj();
            }
            t++;
        }
       
        ans = std::max(ans, h - t + 1);
    }
    std::cout << ans << std::endl;
}

signed main() {
    std::ios::sync_with_stdio(0);
    std::cin.tie(0);
    std::cout.tie(0);

    int t;
    std::cin >> t;
    while (t--) {
        solve();
    }
    return 0;
}
```

##### 求最大平均数子段

题意：给定一个正整数序列，求一个平均数最大的、长度不小于L的（连续的）子段

链接：[Best Cow Fences](https://ac.nowcoder.com/acm/contest/1000/A)

时间复杂度： $O(NlogN)$

思路：通过二分答案转化为对最大子段和的判定

```c++
#include <algorithm>
#include <cmath>
#include <cstdio>
#include <cstring>
#include <iostream>
#include <vector>

int n, l;
const int N = 1e5;
double mapp[N + 2], m[N + 2], s[N + 2];

bool check(double x) {
    for (int i = 1; i <= n; i++) {
        m[i] = mapp[i] - x;
        s[i] = s[i - 1] + m[i];
    }
    double mi = 1e10, ans = -1e10;
    for (int j = l; j <= n; j++) {
        mi = std::min(mi, s[j - l]);
        ans = std::max(ans, s[j] - mi);
    }
    if (ans >= 0)
        return 1;
    else
        return 0;
}

int main() {
    std::cin >> n >> l;
    for (int i = 1; i <= n; i++) {
        std::cin >> mapp[i];
    }

    double esp = 1e-6;
    double l = -1e6, r = 1e6;
    while (r - l > esp) {
        double mid = (l + r) / 2;
        if (check(mid))
            l = mid;
        else
            r = mid;
    }
    std::cout << int(r * 1000) << std::endl;  // l不对。。。不知道why，按理说只是精度问题呀
    // l不对是因为最后的答案是在l，r之间的，但是由于这道题目本身是向下取整，所以用r更为合适，如果这道题目是向上取整的话那么就是用l更为合适
    return 0;
}
```

```c++
        if (k == 0) {
            std::cout << ans << std::endl;
            continue;

            ans = std::max(ans, (n + n % 10) * (k - 1));
        k--;
        n += n % 10;
        if (n % 10 == 0 || !k) {
            std::cout << ans << std::endl;
            continue;
        }
        for (int i = 0; i <= 4; i++) {
            if (!k) break;
            if (i) {
                n += n % 10;
                k--;
            }
            LL l = 0, r = k / 4;
            while (l < r) {
                LL mid1 = l + (r - l) / 3;
                LL mid2 = r - (r - l) / 3;
                if (check(n, mid1, k) > check(n, mid2, k)) {
                    r = mid2 - 1;
                } else
                    l = mid1 + 1;
            }
            ans = std::max(ans, check(n, l, k));
        }
        std::cout << ans << std::endl;
    }
    return 0;
```

```c++
vector<int> Merge(vector<int> a,vector<int> b)
{
	vector<int> ans;
	ans.clear();
	int n=a.size();
	int m=b.size();
	int p_a=0;
	int p_b=0;
	while (p_a<n && p_b<m) {
		if (a[p_a]<=b[p_b]) {
			ans.push_back(a[p_a]);
			p_a++;
			answer+=p_b;
		}
		else {
			ans.push_back(b[p_b]);
			p_b++;
		}
	}

		answer+=(n-p_a)*m;
	}
	if (p_b<m) {
		for (int i=p_b;i<m;i++) ans.push_back(b[i]);
	}
	return ans;
}
```

#### 跳跃的二阶前缀/后缀和

例题：[Sum of Progression](https://codeforces.com/problemset/problem/1921/F)

很巧妙的分快思想，对于d>sqrt(n)的操作直接暴力处理

对于剩下的情况，对于每个d进行O(n)复杂度的预处理

总复杂度 $O_{\sqrt{n}\times n+ \frac{n}{\sqrt{n}}\times n} = O_{n\sqrt n}$ 

```c++
#include <algorithm>
#include <cmath>
#include <cstdio>
#include <cstring>
#include <iostream>

typedef long long ll;
const int N = 2e5;
ll mapp[N + 2], sum[N + 2], sum_2[N + 2], xh[N + 2];
struct group {
    int s, d, k, num;
    bool operator<(const group &t) const {
        return d < t.d;
    }
    ll ans;
} que[N + 2];

bool cmp(group x, group y) {
    return x.num < y.num;
}

int main() {
    int t;
    std::cin >> t;
    while (t--) {
        int n, m;
        std::cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            scanf("%lld", mapp + i);
        }
        for (int i = 1; i <= m; i++) {
            int a, b, c;
            scanf("%d%d%d", &a, &b, &c);
            que[i] = (group){a, b, c, i, 0};
        }
        std::sort(que + 1, que + m + 1);
        for (int i = 1; i <= m; i++) {
            int d = que[i].d, s = que[i].s, k = que[i].k;
            int e = s + (k - 1) * d;

            if ((ll)d * d > n) {
                for (int j = s, ti = 0; j <= e; j += d) {
                    que[i].ans += (++ti) * mapp[j];
                }
            } else {
                if (que[i].d == que[i - 1].d) {
                    que[i].ans = sum_2[s] - sum_2[std::min(e + d, n + 1)] - (xh[s] - xh[e] + 1) * (sum[std::min(e + d, n + 1)]);
                } else {
                    sum[n + 1] = sum_2[n + 1] = xh[n + 1] = 0;
                    for (int i = n; i >= 1; i--) {
                        sum[i] = sum[std::min(n + 1, i + d)] + mapp[i];
                        xh[i] = xh[std::min(n + 1, i + d)] + 1;
                    }
                    for (int i = n; i >= 1; i--) {
                        sum_2[i] = sum_2[std::min(n + 1, i + d)] + sum[i];
                    }
                    que[i].ans = sum_2[s] - sum_2[std::min(e + d, n + 1)] - (xh[s] - xh[e] + 1) * (sum[std::min(e + d, n + 1)]);
                }
            }
        }
        std::sort(que + 1, que + m + 1, cmp);
        for (int i = 1; i <= m; i++) {
            printf("%lld ", que[i].ans);
        }
        puts("");
    }
    return 0;
}
```

#### 普通双指针

**(双指针的根本应用，就是当一个指针固定的时候，另一个指针的和合法位置是按照某一单调顺序排列的，我们就可以用双指针进行更新)** 

```c++
// 这道题也就是判断是否存在双指针可以，事实上是不能把所有等于k的情况遍历完全的，这道题上双指针表现太垃圾了。
// 不服可以参考：https://www.luogu.com.cn/problem/P4149
#include <algorithm>
#include <cmath>
#include <cstdio>
#include <cstring>
#include <iostream>
#include <vector>

const int N = 1e4;
int head[N + 2], nxt[N * 2 + 2], ver[N * 2 + 2], tot, edge[N * 2 + 2];
int q[102], ans[N + 2], v[N + 2], sz[N + 2], mx[N + 2], S, root, d[N + 2], bel[N + 2];
int n, m;
std::vector<int> rem;

void add(int x, int y, int z) {
	tot++;
	edge[tot] = z;
	ver[tot] = y;
	nxt[tot] = head[x];
	head[x] = tot;
}

void find(int x, int fa) {
	sz[x] = 1, mx[x] = 0;
	for (int i = head[x]; i; i = nxt[i]) {
		int y = ver[i];
		if (y == fa || v[y]) continue;
		find(y, x);
		sz[x] += sz[y];
		mx[x] = std::max(mx[x], sz[y]);
	}
	mx[x] = std::max(mx[x], S - sz[x]);
	if (mx[x] < mx[root]) root = x;  // 这里感觉没必要，按理说找到一个就可以返回了，可能是和第一次便利有关吧
}

void getdis(int x, int fa, int dis, int from) {
	rem.push_back(x);
	d[x] = dis;
	bel[x] = from;
	for (int i = head[x]; i; i = nxt[i]) {
		int y = ver[i];
		int z = edge[i];
		if (y == fa || v[y]) continue;
		getdis(y, x, dis + z, from);
	}
}

bool cmp(int x, int y) { return d[x] < d[y]; }
void calc(int x) {
	rem.clear();
	rem.push_back(x);
	d[x] = 0;
	bel[x] = x;  // belong
	for (int i = head[x]; i; i = nxt[i]) {
		int y = ver[i];
		int z = edge[i];
		if (v[y]) continue;
		getdis(y, x, z, y);
	}
	std::sort(rem.begin(), rem.end(), cmp);  // 按照深度排序
	for (int i = 1; i <= m; i++) {
		if (ans[i]) continue;
		int l = 0, r = rem.size() - 1;
		while (l < r) {
			if (d[rem[l]] + d[rem[r]] > q[i])
				r--;
			else if (d[rem[l]] + d[rem[r]] < q[i])
				l++;
			else if (bel[rem[l]] == bel[rem[r]]) {
				if (d[rem[r]] == d[rem[r - 1]])  // 这里一定要这么写，考虑相等的情况，避免相同长度一对不合适就全部没希望，这样就会漏掉可能的情况，l也是一样，只不过正好else就涵盖了
                r--;  
            else
                l++;
			} else {
				ans[i] = 1;
				break;
			}
		}
	}
}

void solve(int x) {
	v[x] = 1;  // 保证了solve遍历不会出现循环
	calc(x);
	for (int i = head[x]; i; i = nxt[i]) {
		int y = ver[i];
		if (v[y]) continue;
		find(y, x);  // 第一次其实就是求sz
		S = sz[y], root = 0, mx[0] = 1e9;
		find(y, x);  // 第二次是找到重心
		solve(root);
	}
}

int main() {
	std::cin >> n >> m;
	for (int i = 1; i < n; i++) {
		int a, b, c;
		std::cin >> a >> b >> c;
		add(a, b, c), add(b, a, c);
	}
	for (int i = 1; i <= m; i++) {
		std::cin >> q[i];
		if (!q[i]) ans[i] = 1;
	}
	solve(1);
	for (int i = 1; i <= m; i++) printf(ans[i] ? "AYE\n" : "NAY\n");
	return 0;
}
```

```c++
#include <algorithm>

typedef long long ll;

ll mapp[N + 2], t[N + 2];

struct group {
    ll x, y;
    bool operator<(const group t) const {
        return x * ans[y] > t.x * ans[t.y];
    }
};
std::priority_queue<group> q;

int main() {
    int n, k;
    std::cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        std::cin >> mapp[i];
    }
    ans.push_back(1);
    for (int i = 1; i <= n; i++) {
        q.push({mapp[i], 0});
    }

    while (k--) {
        // std::cout << ans[ans.size() - 1] << std::endl;
        while (q.top().x * ans[q.top().y] <= ans[ans.size() - 1]) {
            q.push((group){q.top().x, q.top().y + 1});
            q.pop();
        }
        ans.push_back(q.top().x * ans[q.top().y]);
    }
    std::cout << ans[ans.size() - 1] << std::endl;
    return 0;
}
```

博弈论的核心在于必胜！我的理解是，必胜一定是一种必胜的保持，就是每一步我都保证我在必胜态不掉出去，不然对手就一定可以抓住机会反杀。

常见的博弈有如下几个：

**1. 巴什博弈**：有一个数字，两人轮流去取[1,m]的数字，谁不能拿谁输

必胜状态：当数字是(m+1)的倍数的时候就是必输的，其他状态是必胜的

**2. NIM游戏**：有一列数字，每次可以对一个数字取任意值，不能使得该数字变为负数，谁不能拿谁输

必胜状态：抛掉sg函数的繁琐证明，结论就是当这一列数字异或和为0的时候是必输的，其他状态是必胜的

## 分块

分块是一种均摊的思想，是指在面对无法采用有效算法优化的情况下用分类讨论的形式处理不同规模的问题。

> 通常情况下，我们要找到一个问题的性质，这个性质满足数量总和 M 一定。理解这一点很重要，因为这是分块时间复杂度正确的基本保障。这保证了所有大于规模B的问题的数量一定不超过 $\frac{M}{B}$ 。所以找到一个问题的这种性质就是能否使用分块的关键所在。

**例题1：**总操作次数一定

[题目描述](https://atcoder.jp/contests/abc365/tasks/abc365_g)： 有n个人，m条操作，初始情况所有人都是0，每次操作会告诉在某一刻某个人发生了变化，即该人的值异或1。现在有q个询问，每次询问两个人，问这两人都是1的时间的长度总和。

数据规模：$n,m,q\le 2\times10^5,T_i\le 10^9$ 

思考过程：考虑总操作次数m一定，那么我们对操作次数大于B的人和操作次数小于B的人进行分开考虑。对于询问的两个人操作次数小于B的，我们直接暴力操作，即当面对询问的时候，两边对着暴力跳即可。对于询问的至少一个人操作次数大于B的，我们直接对扫描所有操作，暴力统计所有与这个人相关的答案。可以知道操作次数大于B的人数最多有 $\frac{m}{B} $ 个，所以总的时间复杂度就是 $O(qB + \frac{m}{B}*m)$ 所以在 $B = \frac{m}{\sqrt{q}}$ 时可以取得最小值 $ B=m\sqrt{q}$ 

**例题2：**总连边数一定

[题目描述](https://codeforces.com/gym/104065/problem/E)：给一张图，n个点，m条边，有点权有边权。点权转移到相邻节点的单次的花费是点权和边权的乘积。现在给出一个序列A，在i时刻，ai上的点权应该被转移，现在询问整个序列操作结束之后的最小花费。

数据规模：都是1e5吧。。。

思考过程：考虑总边数量是一定的，那么我们根据度进行分类，对于度数大于 B 的单独考虑，对于度数小于 B 的单独考虑。首先考虑去拆分问题，考虑如果我们能知道一开始每个点转移完成整个操作序列所需要的最小花费，那么答案就能被 O(n) 的计算出来。那么这个问题就被拆分成了与点权无关的问题。

然后由于要得到最开始的“需要的最小花费”，所以考虑倒序转移这个方程。$dp[i][j]$ 表示第i个节点完成操作所需要的最小花费（一个有些后缀感觉的状态设计来倒推）1. 当第j次操作与节点i无关的时候 $dp[i][j]=dp[i][j+1]$ 2. 当第j次操作与节点i有关的时候 $dp[i][j]=\min(dp[i][j],dp[k][j+1]+edge[i,k]) ,k\in{adjacent \ of \ vertex \ i}$​ 

观察到这个dp方程大多数情况都是情况1，只有少部分是情况2，所以考虑直接去掉第二维度，dp式的性质没有任何改变。

考虑这个转移的复杂度是O(n)的，这是我们不能接受的。考虑分块。对于度数小于B的节点称为轻节点，对于度数大于B的节点称为重节点。对前者直接暴力处理计算，对后者我们维护一个multiset，每次用最小的去更新。

具体的，在一开始先暴力扫描所有的边，给所有重节点维护一个初始状态。在更新所有节点的时候，更新完自己后。遍历一遍于当前节点相关的所有重节点，去更新一下这个节点。就好了！（考虑整张图上重节点的个数不会超过 $\frac{m}{B}$ 个所以时间正确）

时间复杂度是 $O(q·(B+min(B,\frac{m}{B})·\log n) + q(\log n+ \frac{m}{B} \log n))$ 因为轻节点最多更新B个节点，重节点最多更新所有的重节点，即 $\frac{m}{B}$ 个。$q·(B+\frac{m}{B}\log n)$当 $B=\sqrt{m\log n}$ 时取最小值，为 $O(q\sqrt{m\log n})$ ​ 

**例题3**：总连边数一定的另一种均摊-> 给所有的边定一个方向

题目描述：给定一张图,数三元环的个数。(POJ)

数据范围：$n,m\le 10^5$ 

思考过程：按照度的大小从小到大定义弱序关系建立图，然后暴力枚举，时间复杂度 $O(m\sqrt m)$
